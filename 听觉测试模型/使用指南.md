## 详细使用教程

### 场景1: 批量检测新的音频文件

```python
import joblib
import pandas as pd
import numpy as np
from pathlib import Path
from enhanced_feature_extraction import extract_enhanced_features

# 1. 加载训练好的模型
model_data = joblib.load('optimized_anomaly_model.pkl')
ensemble = model_data['ensemble']
scaler = model_data['scaler']
selected_features = model_data['selected_features']

print(f"使用的特征: {selected_features}")

# 2. 准备要检测的音频文件
audio_folder = Path('.')
audio_files = list(audio_folder.glob('*.wav'))

# 3. 批量检测
results = []
for audio_file in audio_files:
    print(f"检测: {audio_file.name}")

    # 提取特征
    features = extract_enhanced_features(audio_file)

    if features:
        # 准备数据
        df_temp = pd.DataFrame([features])
        X = df_temp[selected_features].fillna(0).values
        X_scaled = scaler.transform(X)

        # 预测
        y_pred, score, pred_dict, score_dict = ensemble.predict(X_scaled)

        # 保存结果
        results.append({
            'filename': audio_file.name,
            'is_abnormal': y_pred[0] == -1,
            'anomaly_score': score[0],
            'confidence': 'High' if score[0] < 0.2 or score[0] > 0.7 else 'Medium'
        })

# 4. 生成报告
results_df = pd.DataFrame(results)
results_df.to_csv('detection_results.csv', index=False, encoding='utf-8-sig')

print("\n检测完成! 结果已保存到 detection_results.csv")
print("\n摘要:")
print(f"总样本数: {len(results_df)}")
print(f"正常样本: {len(results_df[results_df['is_abnormal'] == False])}")
print(f"异常样本: {len(results_df[results_df['is_abnormal'] == True])}")
```

### 场景2: 单个文件实时检测

```python
import joblib
import pandas as pd
from enhanced_feature_extraction import extract_enhanced_features

def detect_audio_anomaly(audio_path, model_path='optimized_anomaly_model.pkl'):
    """
    检测单个音频文件是否异常

    Args:
        audio_path: 音频文件路径
        model_path: 模型文件路径

    Returns:
        dict: 包含检测结果的字典
    """
    # 加载模型
    model_data = joblib.load(model_path)
    ensemble = model_data['ensemble']
    scaler = model_data['scaler']
    selected_features = model_data['selected_features']

    # 提取特征
    features = extract_enhanced_features(audio_path)

    if not features:
        return {'error': 'Failed to extract features'}

    # 准备数据
    df = pd.DataFrame([features])
    X = df[selected_features].fillna(0).values
    X_scaled = scaler.transform(X)

    # 预测
    y_pred, score, pred_dict, score_dict = ensemble.predict(X_scaled)

    # 分析结果
    is_abnormal = y_pred[0] == -1
    anomaly_score = score[0]

    # 判断置信度
    if anomaly_score < 0.2:
        confidence = 'Very High'
        status = '严重异常'
    elif anomaly_score < 0.4:
        confidence = 'High'
        status = '中度异常'
    elif anomaly_score < 0.6:
        confidence = 'Medium'
        status = '可疑'
    elif anomaly_score < 0.8:
        confidence = 'High'
        status = '基本正常'
    else:
        confidence = 'Very High'
        status = '完全正常'

    return {
        'filename': Path(audio_path).name,
        'is_abnormal': is_abnormal,
        'anomaly_score': anomaly_score,
        'status': status,
        'confidence': confidence,
        'individual_predictions': {k: v[0] == -1 for k, v in pred_dict.items()},
        'individual_scores': {k: v[0] for k, v in score_dict.items()}
    }

# 使用示例
result = detect_audio_anomaly('test_audio.wav')
print(f"文件: {result['filename']}")
print(f"状态: {result['status']}")
print(f"异常得分: {result['anomaly_score']:.4f}")
print(f"置信度: {result['confidence']}")
print(f"是否异常: {'是' if result['is_abnormal'] else '否'}")
```

### 场景4: 调整检测灵敏度

```python
import pandas as pd
from optimized_anomaly_model import EnsembleAnomalyDetector, intelligent_feature_selection
from sklearn.preprocessing import RobustScaler
import joblib

# 加载数据
df = pd.read_csv('enhanced_audio_features.csv')

# 特征选择
try:
    disc_df = pd.read_csv('discrimination_analysis.csv')
except:
    disc_df = None

selected_features = intelligent_feature_selection(df, disc_df, method='hybrid', top_n=15)

# 准备数据
X = df[selected_features].fillna(0).values
y = df['is_abnormal']
scaler = RobustScaler()
X_scaled = scaler.fit_transform(X)

# 训练集
normal_mask = df['is_abnormal'] == False
X_train = X_scaled[normal_mask]

# 尝试不同的contamination值
contamination_values = [0.2, 0.35, 0.45, 0.55, 0.7]

for contam in contamination_values:
    print(f"\n{'='*60}")
    print(f"Contamination = {contam}")
    print('='*60)

    # 训练模型
    ensemble = EnsembleAnomalyDetector(contamination=contam)
    ensemble.fit(X_train, feature_names=selected_features)
    ensemble.scaler = scaler
    ensemble.selected_features = selected_features

    # 预测
    y_pred, scores, _, _ = ensemble.predict(X_scaled)
    y_pred_bool = (y_pred == -1)

    # 评估
    from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

    print(f"准确率: {accuracy_score(y, y_pred_bool):.4f}")
    print(f"精确率: {precision_score(y, y_pred_bool, zero_division=0):.4f}")
    print(f"召回率: {recall_score(y, y_pred_bool, zero_division=0):.4f}")
    print(f"F1分数: {f1_score(y, y_pred_bool, zero_division=0):.4f}")

    # 保存最佳模型
    if f1_score(y, y_pred_bool, zero_division=0) >= 0.95:
        joblib.dump({
            'ensemble': ensemble,
            'scaler': scaler,
            'selected_features': selected_features,
            'contamination': contam
        }, f'model_contam_{contam:.2f}.pkl')
        print(f"✓ 模型已保存为 model_contam_{contam:.2f}.pkl")
```

## 常见问题

### Q1: 模型训练失败，报错"样本数不足"

**A**: 某些模型(如EllipticEnvelope)需要足够的样本数。优化后的代码已经自动处理这个问题，会根据样本数动态选择模型。

### Q2: 如何提高检测准确率？

**A**:
1. 收集更多样本数据
2. 尝试不同的特征选择方法
3. 调整contamination参数
4. 使用数据增强技术

### Q3: 如何解释异常得分？

**A**:
- **< 0.2**: 高置信度异常，需要立即检查
- **0.2-0.4**: 中度异常，建议检查
- **0.4-0.6**: 可疑区域，需要进一步观察
- **0.6-0.8**: 基本正常
- **> 0.8**: 高置信度正常

### Q4: 可以只使用基础特征吗？

**A**: 可以。如果`enhanced_audio_features.csv`不存在，模型会自动使用`audio_features.csv`。但增强特征通常能提供更好的性能。

### Q5: 如何处理新类型的异常？

**A**:
1. 收集新异常样本
2. 标记为异常 (is_abnormal=True)
3. 重新提取特征
4. 重新训练模型

### Q6: 模型可以部署到生产环境吗？

**A**: 可以。模型保存为.pkl文件，可以轻松加载和使用。建议:
1. 定期用新数据重新训练
2. 监控模型性能指标
3. 建立人工审核机制

## 性能优化建议

### 1. 加速特征提取

```python
# 使用多进程并行处理
from multiprocessing import Pool
from pathlib import Path

def extract_features_parallel(audio_files, n_processes=4):
    with Pool(n_processes) as pool:
        results = pool.map(extract_enhanced_features, audio_files)
    return [r for r in results if r is not None]

audio_files = list(Path('.').glob('*.wav'))
features_list = extract_features_parallel(audio_files)
```

### 2. 减少内存占用

```python
# 分批处理大量文件
def batch_predict(audio_files, model_path, batch_size=100):
    model_data = joblib.load(model_path)

    for i in range(0, len(audio_files), batch_size):
        batch = audio_files[i:i+batch_size]
        # 处理当前批次
        results = []
        for audio_file in batch:
            result = detect_audio_anomaly(audio_file, model_path)
            results.append(result)

        # 保存批次结果
        pd.DataFrame(results).to_csv(
            f'results_batch_{i//batch_size}.csv',
            index=False
        )
```


## 集成到应用系统

### Web API 示例 (Flask)

```python
from flask import Flask, request, jsonify
import joblib
from enhanced_feature_extraction import extract_enhanced_features
import tempfile
import os

app = Flask(__name__)

# 加载模型
model_data = joblib.load('optimized_anomaly_model.pkl')

@app.route('/detect', methods=['POST'])
def detect():
    if 'audio' not in request.files:
        return jsonify({'error': 'No audio file provided'}), 400

    audio_file = request.files['audio']

    # 保存临时文件
    with tempfile.NamedTemporaryFile(delete=False, suffix='.wav') as temp:
        audio_file.save(temp.name)
        temp_path = temp.name

    try:
        # 提取特征并预测
        result = detect_audio_anomaly(temp_path)
        return jsonify(result)
    finally:
        # 清理临时文件
        os.unlink(temp_path)

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
```

### 命令行工具

```python
#!/usr/bin/env python3
import argparse
import sys
from pathlib import Path

def main():
    parser = argparse.ArgumentParser(description='音频异常检测工具')
    parser.add_argument('audio_file', help='待检测的音频文件')
    parser.add_argument('--model', default='optimized_anomaly_model.pkl',
                       help='模型文件路径')
    parser.add_argument('--threshold', type=float, default=0.5,
                       help='异常判定阈值 (0-1)')

    args = parser.parse_args()

    if not Path(args.audio_file).exists():
        print(f"错误: 文件 {args.audio_file} 不存在", file=sys.stderr)
        sys.exit(1)

    result = detect_audio_anomaly(args.audio_file, args.model)

    print(f"文件: {result['filename']}")
    print(f"状态: {result['status']}")
    print(f"异常得分: {result['anomaly_score']:.4f}")

    if result['is_abnormal']:
        print("结论: ⚠️ 检测到异常")
        sys.exit(1)
    else:
        print("结论: ✓ 正常")
        sys.exit(0)

if __name__ == '__main__':
    main()
```
